## اکھ یُتھ آغاز یَمِک مقاْم ماخذ پختہٕ پٲٹھۍ نہٕ ہاوان

چندر گپت زایہِ ژۄرمہِ صدی قبل مسیح مَنٛز، مَگر نہٕ تَہٕنٛز صٔحیح تاریخِ پیدٲئش تہٕ نہٕ جائے پیدٲئش چھِ پختہٕ پٲٹھۍ ثٲبِت۔ پَتہٕ چِہ بیانات چھِ تِمن مُختلِف خاندانی پَس منظرَس سٟتۍ جوڈان۔ تٲریخی نُکتہٕ آغاز چھُ اکھ یُتھ نوجوان یَمۍ بادشاہت حٲصِل کٔر، نہٕ زِ اکھ یُتھ بَچپَن یَمِک پرٛٮ۪تھ مَنظر از سَرِ نو تَشکیٖل دِتھ ہٮ۪کوٗ۔ [1, 4, 5]

## واریاہن روایاتَو کٔر تَہٕنٛدِس آغازَس پؠٹھ دعویٰ

بودھ تہٕ جین لؠکھارٮ۪و کٔر خاندان کِس بانی یَس مُختلِف شجرہٕ نسب تہٕ مذہبی پَس منظر کِس وسِیلہٕ یاد۔ تَہٕنٛزٕ کہٲنی چھِ اَمِک ثبوٗت زِ مُختلِف بَرادرِیہِ ٲسۍ تَہٕنٛزِ اہمیتَس کِتھ کٔنۍ سَمجان۔ تِم پَزَن نہٕ اَکھ جاے کٔرِتھ کانہہ ظٲہری ہم عصر خاندانی رجسٹر بَناونہٕ ین، تہٕ نہٕ تَہٕنٛدِس سمٲجی پَس منظر کٮ۪ن مُتنازعہٕ دعوَن غلط یقیٖنس سٟتۍ طَے کَرنہٕ خٲطرٕ اِستعمال کَرنہٕ یِن۔ [4, 5]

## وادیِ گنگا نِش دوٗر اکھ بدلَن واجِنۍ دُنیا

سکندر سٕنٛدِ حملَن تہٕ تَہٕنٛزِ مَوتہِ کٔر چندر گپت سٕنٛزِ سلطنت قٲئم گژھنہٕ برٛونٛہہ شُمال مغرب اِنتشارس مَنٛز مُبتلا۔ یونٲنی بَستِیہِ، مقٲمی حُکمران، تہٕ جانشین رِیاستَن ٲس تَتھ دُنیاہُک حِصہٕ یَمِک سامنہٕ چندر گپتَس کَرُن اوس۔ موریہ سُنٛد پَتہٕ چُک کارنامہٕ اوس نہٕ مَحض سکندر سٕنٛزن فتوحاتن ہُنٛد تسلسُل: اَمِہ کٔر ہندوستانَس مَنٛز اکھ نۆو حُکمرانی مَرکَز پٲدا۔ [2, 3, 6]

## سُہ مشیٖر یُس تَہٕنٛزِ ذاتہِ سٟتۍ لازم و ملزوٗم بنیو

ہندوستٲنی روایات چھِ چندر گپت سٕنٛدِس عروٗجَس **چانکیہ** سٟتۍ جوڈان، یِمن کوٹلیہ تہِ ونان چھِ۔ یہِ شراکَتھ بَنیے ذَہانت، عزاَئم تہٕ اَکہِ خاندان کِس خٲتمٕچ اکھ پُراثر دٲستان۔ اَمِچ یاد تھاونہٕ آمٕژ اہمیت چھِ صَاف؛ پَتہٕ چٮ۪ن روایاتن مَنٛز نِجی گُفتگوٗ تہٕ ہوشیارانہٕ حِکمت عملی چھِ نہٕ تَہٕنٛزِ منصوٗبہٕ بندی ہٕنٛز ہم عصر تحریرٕ۔ [1, 4, 5]

## سُہ شاہی خاندان یَمِک تِمو تَختہٕ پلٹوو

موریہ نِش برٛونٛہہ اوس نند خاندان مگدھٕچ طاقتوَر سلطنتَس پؠٹھ قٲبِض۔ چندر گپت سٕنٛزِ کامِیٲبی مَنٛز اوس تِہُنٛد خاتمہٕ تہٕ پَننہِ بادشاہتُک قِیام شٲمِل۔ بَچِمٕژ روایات چھِ نہٕ اَتھ نَتیجس تام وٲتنس پؠٹھ مُتفِق۔ محلاتِی سازشن تہٕ لڑٲین ہُنٛد اکھ صَاف سُتھرٕ تسلسُل دَر حَقیقت چھُ اَتھ کَتھ چھُپاون زِ شواہد کٔتیا ناہموار چھِ۔ [4–6]

## اکھ نۆو شاہی گھروار آو اِقتدارَس مَنٛز

روایتی طور پؠٹھ چھِ چندر گپت سٕنٛز تخت نشینی تقریباً **322 قبل مسیح** ماننہٕ یِوان۔ یہِ تٲریخ چھِ اکھ بازیافت، کانہہ محفوٗظ تاج پوشی ہُنٛد پروانہٕ نہٕ۔ یۄس کَتھ پختہٕ پٲٹھۍ اَہَم چھِ سۄ چھِ سِیاسی تبدیٖلی: اَکہِ نٔوِ خاندانَن سنبھالو گنگا ہٕنٛزِ سلطنتُک اِقتدار تہٕ تَہٕنٛز اِختیار پھیلاوو تَتھ خِطہٕ نِش واریاہ دوٗر یَتھ مَنٛز اَتھ طاقتہِ ہٕنٛزِ جَڑٕ ٲس۔ [4, 6]

## دٔریاوَن ہٕنٛدٮ۪ن بٔٹِھین پؠٹھٕ چَلاونہٕ یِن واجِنۍ سلطنت

جدید پٹنہ کِس نزدیٖک واقع **پاٹلی پُتر** بنیو موریہ سُنٛد دارالحکومت۔ دٔریٲیی واتَن کٔر اَمِک دربار زرخیٖز علاقَن تہٕ دوٗر دَراز بَستؠن سٟتۍ جوڈِتھ۔ اَکہِ سلطنتَس بَرقرار تھَونس خٲطرٕ ٲس فوجی فتوحات سٟتۍ سٟتۍ معلوٗمات، وسٲیل تہٕ لۆکن ہٕنٛز نَقل و حَرکت تہِ ضروٗری۔ بادشاہُک اِقتدار اوس تِمَن اِدارَن تہٕ برادرین پؠٹھ مُنحصر یِمن وسِیلہٕ یہِ نَقل و حَرکت مُمکِن گژھِتھ ہٮ۪کہِ ہٚے ۔ [1, 6, 7]

## اَکہِ نٔوِ پیمانس پؠٹھ فتوٗحات

کلاسیکی بیانات چھِ چندر گپتَس اَکہِ انتہٲیی وسیع علاقس کِس حُکمران پٲٹھۍ یاد کَران۔ تَہٕنٛز جامع تفصیلات پَزِ نہٕ اَتھ دعوٲیس مَنٛز بدلاوُن زِ جدید جَنوٗبی ایشیا کِس پرٛٮ۪تھ حِصَس پؠٹھ ٲس اَکوی پٲٹھۍ حُکوٗمت کَرنہٕ یِوان۔ اَتھ باوُجوٗد چھِ تِم ظٲہر کَران زِ تَہٕنٛدِس عروٗجَس آو ہندوستان نِش نؠبر طاقت کِس توازنس مَنٛز اَکہِ بٔڑِ تبدیٖلی پٲٹھۍ تسلیٖم کَرنہٕ۔ [2, 3, 6]

## بادشاہت کِس پٔتھِمہِ طرفہٕ کھڑا فوج

پلوٹارکَن منسوٗب کٔر چندر گپتَس سٟتۍ اکھ واریاہ بٔڑ فوج۔ یِتھ اعداد و شُمار چھِ اَدبی دعویٰ، کانہہ تصدیق شُدٕ فوجی گوشوارٕ نہٕ۔ بُنیادی نُکتہٕ چھُ یہِ زِ نٔوِ بادشاہتَس نِش ٲسۍ وسیع وسٲیل۔ سِپاہی، ۂسۍ، رَسَد، تہٕ تِہُنٛز مَدَتھ کَرن وٲلۍ لۄکھ ٲسۍ تَتھ نِظامُک حِصہٕ یَمۍ سامراجی توسیع مُمکِن بَنٲو۔ [2]

## سلوقس آو مَشرِق کُن

سکندر سٕنٛدٮ۪ن جانشینَن منٛزٕ اکھ، **سلوقس اول**، پَکیو ہندوستان کُن۔ چندر گپتَس سٟتۍ تَہٕنٛد ٹکراو گوٚو موریہ سلطنتَس پؠٹھ مُستقِل سلوقی فَتح کِس بَدلہٕ اَکہِ مُعاہدَس پؠٹھ خَتٕم۔ مُہماتن ہٕنٛز صٔحیح نَقل و حَرکت چھِ واریاہ کَم دَرج؛ پرٛٮ۪تھ معرکٕچ کُنہِ تفصیٖلی جدید تشکیٖل کِس مُقابلَس مَنٛز چھُ یہِ مُعاہدٕ زِیادٕ مُستند پٲٹھۍ ثٲبِت۔ [3, 6]

## اکھ یُتھ مُعاہدہٕ یَمۍ سرحد بدلوو

اسٹرابو چھُ بیَان کَران زِ سلوقسن ترٛٲو ہَستؠن تہٕ اَکہِ ازدواجی رِشتہٕ پؠٹھ مُشتمل اَکہِ مُعاہدٕ تحت چندر گپتَس علاقہٕ مُنتقِل۔ جغرافیٲیی وسعت تہٕ اَتھ رِشتہٕ کِس مفہوٗمَس چھُ اِحتیاط سٟتۍ سمجنٕچ ضروٗرت۔ یہِ اِقتباس چھُ اَکہِ اَہَم سَرحدی مُعاہدٕچ تصدیٖق کَران؛ یہِ چھُ نہٕ جدید سَرحدن ہُنٛد کانہہ مُکمل نقشہٕ یا کُنہِ نامزَد خاندٲنی کۄڑِ ہٕنٛز سَوانِح فراہم کَران۔ [3]

## تٲریخی بیاناتن مَنٛز پانژھ ہَتھ ۂسۍ

چندر گپت تہٕ سلوقس کِس بارَس مَنٛز کلاسیکی شہادتَن مَنٛز چھِ **پانژھ ہَتھ ہَستؠن** ہٕنٛز تعداد سامنے یِوان۔ یہِ چھِ اَتھ لین دینَس سٟتۍ منسوٗب پیمانہٕ تہٕ ہیلینسٹک سِیاستَس مَنٛز ہندوستٲنی جنگی ہَستؠن ہٕنٛز اہمیت واضح کَران۔ یہِ بیَان چھُ قدیم ماخذَن ہُنٛد حِصہٕ؛ یہِ چِھنہٕ دۄشوٕنی دربارن ہٕنٛدِ طرفہٕ دَستخط شُدٕ کانہہ محفوٗظ فِہرِست۔ [2, 3]

## ”ہیلینا“ سٕنٛز کانہہ پختہٕ شِناخت نہٕ

عوٲمی کہٲنؠن مَنٛز چھُ بعض اوقات سلوقس سٕنٛزِ کوٗرِ چندر گپت سٕنٛزِ زنانہِ پٲٹھۍ پیش کَرنہٕ یِوان تہٕ تَہٕنٛدِس گِرد اکھ تفصیٖلی رومانوی دٲستان گھڑنہٕ یِوان۔ مُعاہدٕچ بَچِمٕژ شہادت چھِ نہٕ کُنہِ مَخصوٗص ناوٕ والہِ شخصیَتھ پختہٕ پٲٹھۍ ثٲبِت کَران۔ کُنہِ ازدواجی تعلُق یا خاندانہِ مُتعلِق مُعاہدٕ ہٮ۪کہِ نہٕ پانہٕ کُنہِ زنانہِ ہُنٛد ناو، نِجی زِندگی، یا موریہ دربارَس تام تَہٕنٛد سَفَر سامنے اَنِتھ۔ [3, 6]

## بیرونی مُبصِر ؤتۍ دربارَس تام

**میگاستھینز** بنیو سلوقی سَفیٖر سٕنٛزِ حِیثیت سٟتۍ چندر گپت سٕنٛدِس دربارَس سٟتۍ وابستہٕ۔ تَہٕنٛد بیانات چھِ کُنہِ مُکمل اصل خَطی نُسخہٕ کِس بَدلہٕ پَتہٕ چٮ۪ن کلاسیکی لؠکھارٮ۪ن ہٕنٛدٮ۪ن اِقتباساتن تہٕ خُلاصَن مَنٛز محفوٗظ۔ یہِ ترسیٖل چھِ موریہ دُنیاہُک اکھ اَہَم مَنظر پیش کَران، مَگر یہِ چھِ مُبصِر تہٕ اَز کِس پَرَن وٲلِس درمِیان وِچولٮ۪ن تہِ اَنان۔ [6, 7]

## کُنہِ نیٖبَرِمِس مُبصِرٕچ درجہٕ بندی چھِ نہٕ مردم شُمٲری

یونٲنی بیانات چھِ ہندوستٲنی شَہَرن، سمٲجی گروہَن، رَسم و رواجَن تہٕ شاہی اِنتظاماتن پؠٹھ بَحث کَران۔ تِم چھِ پَننٮ۪ن لؠکھارٮ۪ن ہٕنٛزِ دِلچسپی تہٕ توقُعاتن ہٕنٛز عکاسی کَران قِیٖمتی شہادت پیش کَران۔ تَہٕنٛزَن تفصیلاتن پَزِ باقٕے شواہد سٟتۍ مُوازنہٕ کَرنہٕ یُن، بَجاے اَمِکہِ زِ تِمن اکھ یکساں سَروے سمجنہٕ یِیہِ یَمِچ پرٛٮ۪تھ تفصیٖل چندر گپت سٕنٛزِ سلطنت کٮ۪ن سٲرنی خطَن پؠٹھ لاگوٗ گژھان آسِہ۔ [6, 7]

## ارتھ شاستر تہٕ تٲریخی بادشاہ

**ارتھ شاستر** چھُ کوٹلیہ تہٕ چندر گپت سٕنٛدِس عروٗجٕچ روایَتہِ سٟتۍ جوڈنہٕ یِوان۔ اَمِک محفوٗظ مَتن چھُ تصنیٖف تہٕ تَرسیٖلٕچ اکھ پیچیدٕ تٲریخ تھَوان۔ یہِ چھُ پَرَن والٮ۪ن حُکوٗمت، سُراغ رسٲیی، محاصِل تہٕ شاہی تَحفُظ کٮ۪ن نظِریاتن ہٕنٛز چھان بیٖن کَرنس مَنٛز مَدَتھ دِوان؛ یہِ چِھنہٕ کانہہ یِتھ روزنامچہٕ ڈائری یۄس یہِ ثٲبِت کَرِ زِ چندر گپتَن کٔر پرٛٮ۪تھ ہِدایتہِ پؠٹھ ذٲتی طور پؠٹھ عمل۔ [1, 6]

## واریاہ پَتہٕ چہِ اَکہِ کہٲنی مَنٛز دردھرا

**ہیم چندر** سُنٛد محفوٗظ کَرنہٕ آمُت جین بیَان چھُ چندر گپت سٕنٛدِس نٮ۪چوٗ بندوسار سٕنٛزِ ماجہِ پٲٹھۍ **دردھرا** سُنٛد ناو دِوان۔ یہِ چھُ تَہٕنٛزِ مَوتَس زَہَر آلوٗد شاہی کھٮ۪نَس تہٕ چانکیہ سٕنٛدِ طرفہٕ شُرِس ہٕنٛزِ ہنگامی نَجات سٟتۍ جوڈان۔ یہِ چھِ پَتہٕ چہِ اکھ پُراثر دٲستان، نہٕ زِ موریہ پیدٲئشہِ بارَس مَنٛز کانہہ ہم عصر طِبی ثبوٗت۔ [1, 5]

## خوفناک نَتائجَن واجِنۍ اکھ حِفاظتی تدبیٖر

اَتھ کہٲنی مَنٛز ٲس چندر گپت سٕنٛدِس کھٮ۪نَس مَنٛز مدافعت پٲدا کَرنہٕ خٲطرٕ لۄکژِ مقدارس مَنٛز زَہَر ملاونہٕ یِوان۔ دردھرا چھِ انجانَس مَنٛز اَتھ کھٮ۪نَس مَنٛز شَریٖک گژھان۔ یہِ کہٲنی چھِ چھُپِمٕتِس عِلم کِس وسِیلہٕ کُنہِ حُکمران سٕنٛزِ حِفاظتہِ سٟتۍ پٲدا گژھن وٲلِس خطرس پرکھان۔ اَمِچ اَخلاقی تہٕ اَدبی قُوّت پَزِ اَتھ کَتھ کِس تصدیق شُدٕ بیَان نِش اَلگ روزُن زِ شاہی خاندانَن کِیاہ کھٮ۪ومُت اوس۔ [1, 5]

## بندوسارَن جوڈٕ نَسلہِ

**بندوسار** بَنِیہِ چندر گپت سٕنٛدۍ جانشین تہٕ اشوک سٕنٛدۍ مول۔ اَتھ جانشینی کٔر موریہ سٕنٛزِ بُنیاد اَکہِ شَخص کِس دورِ حُکوٗمتہٕ کھۄتہٕ بَڈِتھ قٲئم۔ پَتہٕ چِہ توٲریخ چھِ اَتھ تسلسُلَس صَاف پٲٹھۍ محفوٗظ تھَوان، حالانکہ تَہٕنٛدٮ۪ن ادوارَن ہٕنٛز طوالت تہٕ قَطعی تٲریخن ہٮ۪کوٗ نہٕ سَہل پٲٹھۍ کُنہِ اَکوی غٲر مُتنازعہٕ کیلنڈرس مَنٛز یکجا کٔرِتھ۔ [1, 4]

## سُہ سَفَر یُس زِندگی کٮ۪ن ٲخری دۄہَن سٟتۍ یاد تھَونہٕ آو

پَتہٕ چہِ جین روایات چھِ چندر گپت سٕنٛدِس تَرکِ دُنیاہَس **بھدراباہو** تہٕ **شراون بیلگولہ** کِس مذہبی مَنظر نامَس سٟتۍ جوڈان۔ یِم روایات چھِ تَہٕنٛزِ یاد تھاونہٕ آمٕژ مذہبی زِندگی خٲطرٕ مَرکزی حِیثیت تھَوان۔ تِہُنٛد پَتہٕ قلمبند گژھُن تہٕ مُختلِف بیاناتن مَنٛز شَخصیاتن ہٕنٛز شِناخت چھِ تِمن شہنشاہ سٕنٛدِس ٲخری سَفَر کِس ہم عصر سَفَر نامَس پٲٹھۍ کٲم کَرنہٕ نِش رۆکاوان۔ [5, 6]

## بسترِ مرگ کِس ہم عصر رِکارڈٕ وَرٲے اکھ انجام

روایتی طور پؠٹھ چھُ چندر گپت سُنٛد دورِ حُکوٗمت تقریباً **297 قبل مسیح** کِس نزدیٖک خَتٕم گژھان۔ جین روایَتھ چھِ فاقہٕ کَشی کِس وسِیلہٕ اَکہِ زاہدانہٕ موتَس یاد کَران، مَگر بَچِمُت کانہہ ہم عصر بیَان چھُ نہٕ قَطعی حالات ثٲبِت کَران۔ اَتھ روایَتہِ احتِرام سٟتۍ بیَان کَرُن چھُ مُمکِن تہٕ اَتھ سٟتۍ سٟتۍ یاد تھاونہٕ آمٕتِس روٗحانی انجامَس زِیادٕ یقیٖنی طور قٲئم شُدٕ خاندانی جانشینی نِش اَلگ تھَوُن تہِ مُمکِن چھُ۔ [4–6]

## سۄ بُنیاد یۄس جانشینَن وِرثَس مَنٛز مِلِیہِ

بندوسارَس تہٕ پَتہٕ اشوکَس مِلیے اکھ یِتھ بادشاہت وِرثَس مَنٛز یۄس برٛونٛہہ یے سامراجی پیمانس پؠٹھ کٲم کَران ٲس۔ تَہٕنٛدٮ۪ن ادوارَن پَزِ نہٕ پوٗرٕ پٲٹھۍ پَتھ کُن نِتھ چندر گپتَس پؠٹھ لاگوٗ کَرُن۔ تَہٕنٛد کارنامہٕ اوس بُنیاد ترٛاوُن: اَکہِ پرٛانہِ حُکمران خاندانُک خاتمہٕ، اَکہِ نٔوِ خاندانٕچ تَوسیع، تہٕ اَکہِ بٔڑِ ہیلینسٹک طاقتہِ سٟتۍ تسلیٖم شُدٕ سَرحدی تعلُقات۔ [1–4, 6]

## بانی بَنیو واریاہن تٲریخن ہُنٛد کِردار

چندر گپت روٗدِ کلاسیکی تٲریخن، بودھ تواریخن، جین روایاتن، سِیاسی اَدَب تہٕ پَتہٕ کٮ۪ن فَنَن مَنٛز زِندہٕ۔ یِم ماخذ چھِ مُختلِف قٕسمٕچ معلوٗمات تہٕ بادشاہت کِہ مُختلِف تصوٗرات محفوٗظ تھَوان۔ تِمن اَکھ جاے پَرُن چھُ تَہٕنٛزِ زِندگی زِیادٕ پُرمایہِ بَناوان؛ پرٛٮ۪تھ واقعَس یکساں پٲٹھۍ ہم عصر سمجُن چھُ اَتھ کَم قابِلِ اِعتماد بَناوان۔ [1–7]

## تَہٕنٛدِس دورَن کِیاہ بَدلوو

چندر گپتَن بدلوو شُمالی ہندوستانُک سِیاسی پیمانہٕ تہٕ ترٛٲو تَتھ خاندانٕچ بُنیاد یُس بندوسار تہٕ اشوک سٕنٛدِ وسِیلہٕ حُکوٗمت کَرِ۔ غٲر یقیٖنی بَچپَن تہٕ پَتہٕ چہِ مذہبی روایات چھِ پننٮ۪ن صَاف حَدَن سٟتۍ تَہٕنٛزِ دٲستانُک حِصہٕ بنِتھ روزان۔ تَہٕنٛزِ اہمیتَس خٲطرٕ چِھنہٕ خالِی جاے بَرنہٕ باپَتھ کُنہِ پوٗرٕ من گھڑت نِجی زِندگی ہٕنٛز ضروٗرت۔ [1–7]
